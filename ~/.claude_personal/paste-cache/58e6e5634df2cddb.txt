
// =====================
// PROFILES (OAuth users)
// =====================
Table profiles {
  id uuid [primary key, note: "matches auth.users.id"]

  username varchar [unique, not null]
  email varchar [unique, not null]

  full_name varchar
  avatar_url text

  created_at timestamp
  updated_at timestamp
}

// =====================
// PERIODS (Month or Trip)
// =====================
Table periods {
  id integer [primary key]

  user_id uuid

  name varchar
  type varchar

  year integer
  month integer

  start_date date
  end_date date

  total_budget decimal

  created_at timestamp
}

// =====================
// CATEGORIES
// =====================
Table categories {
  id integer [primary key]

  user_id uuid
  name varchar

  created_at timestamp
}

// =====================
// EXPENSES
// =====================
Table expenses {
  id integer [primary key]

  period_id integer
  category_id integer

  amount decimal
  description varchar
  spent_at date

  created_at timestamp
}

Ref: profiles.id < periods.user_id
Ref: profiles.id < categories.user_id

Ref: periods.id < expenses.period_id
Ref: categories.id < expenses.category_id
